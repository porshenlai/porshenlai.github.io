輸入防線

  • [PII 去敏與攻擊偵測] (攔截越獄/隱藏身分證)
  • # =========================================================================== # 防線 1：輸入端防火牆 (Input Guardrail) - PII 遮蔽與惡意 Prompt 攔截 # =========================================================================== def input_guardrail(user_input: str) -> str: print("\n🛡️ [第 1 道防線：輸入端檢查]") # A. 檢測越獄/惡意攻擊關鍵詞 (Jailbreak / Prompt Injection) jailbreak_keywords = ["IGNORE PREVIOUS INSTRUCTIONS", "DAN", "忽略先前的指令", "教我製造", "內線交易"] for kw in jailbreak_keywords: if kw.lower() in user_input.lower(): raise ValueError(f"❌ 警告：檢測到惡意輸入或越獄企圖！(關鍵字: {kw})") # B. 個人敏感資料 (PII) 自動去敏 (如：台灣身分證字號 Regex 遮蔽) taiwan_id_regex = r"[A-Z][12]\d{8}" sanitized_input = re.sub(taiwan_id_regex, "[身分證已遮蔽]", user_input) if sanitized_input != user_input: print(" ⚠️ 偵測到 PII 身分證字號，已完成去敏處理。") print(" ✅ 輸入安全檢查通過。") return sanitized_input

RAG 防線

  • [權限過濾 (Metadata)] (隔離未授權文件)
  • # =========================================================================== # 防線 2：RAG 資料存取權限控管 (Retrieval Guardrail) # =========================================================================== def setup_rag_with_rbac(): # 建立測試文件並帶有不同權限標籤 (Metadata) embeddings = HuggingFaceEmbeddings(model_name="BAAI/bge-small-zh-v1.5") docs = [ {"text": "台積電 2026 年 Q2 財報：營業收入達 6,500 億元。", "role": "public"}, {"text": "機密：A客戶授信評等為低，存在高違約風險，身份證A123456789。", "role": "admin"} ] vector_store = Chroma(collection_name="safety_test", embedding_function=embeddings) # 寫入向量庫 from langchain_core.documents import Document doc_objects = [Document(page_content=d["text"], metadata={"role": d["role"]}) for d in docs] vector_store.add_documents(doc_objects) return vector_store def rbac_retrieval(vector_store, query: str, user_role: str = "user"): print("\n🛡️ [第 2 道防線：RAG 權限過濾]") # 根據使用者角色限定搜尋範圍 (RBAC) retriever = vector_store.as_retriever( search_kwargs={ "k": 2, "filter": {"role": "public"} if user_role == "user" else {} } ) docs = retriever.invoke(query) print(f" 🔍 使用者權限 [{user_role}]，成功撈取 {len(docs)} 筆授權資料。") return "\n".join([d.page_content for d in docs])

模型防線

  • [System Prompt 約束] (Gemma 內建拒絕機制)
  • # =========================================================================== # 防線 3：模型與 System Prompt 約束 (Model Safety Alignment) # =========================================================================== def call_gemma_model(context: str, query: str) -> str: print("\n🛡️ [第 3 道防線：地端 Gemma 模型推論]") llm = Ollama(model="gemma2:2b", temperature=0.1) system_prompt = f"""你是一位合規的金融 AI 助手。 請嚴格遵守以下限制： 1. 僅根據【參考資料】回答，若無資料請回答「無法提供」。 2. 絕對禁止給出「保證獲利」、「全壓買進」等非法投資承諾。 【參考資料】： {context} 【使用者提問】： {query} 【回答】：""" response = llm.invoke(system_prompt) print(" ✅ Gemma 模型生成完成。") return response

輸出防線

  • [合規校驗與敏感詞過濾] (修飾金融過度承諾)
  • # =========================================================================== # 防線 4：輸出端過濾與合規修飾 (Output Guardrail) # =========================================================================== def output_guardrail(raw_response: str) -> str: print("\n🛡️ [第 4 道防線：輸出端合規檢查]") # 檢查是否包含金融禁言 (如保證獲利、保證保本等過度承諾字眼) forbidden_words = ["保證獲利", "絕對賺錢", "100%獲利", "全壓買進", "保證不虧"] sanitized_response = raw_response for word in forbidden_words: if word in sanitized_response: print(f" ⚠️ 警示：輸出的內容包含金融違規詞彙 [{word}]，已強制進行免責修飾！") sanitized_response = sanitized_response.replace(word, "（警語：投資存在風險，請獨立評估）") # 加上強制免責聲明 sanitized_response += "\n\n【免責聲明】：本回答由地端 AI 系統生成，僅供參考，不構成任何投資決策依據。" print(" ✅ 輸出端安全審查完成。") return sanitized_response
